\subsection{(H2) at $L=13$ \status{certified (interval arithmetic, outward rounding; verified)}}\label{sec:certH2}

Sources: \file{H1_ANALYTIC.md} \S0--\S8 (statement, lemmas, majorant); verifier \file{H1_ANALYTIC_VERIFY.md} V0--V9 with \file{numerics/h1analytic/bb_arb.py}; operator check \file{H1_ANALYTIC_VERIFY2.md} with \file{numerics/h1analytic/indep_H.py}; symmetry check \file{H2_CERTIFY_P5_VERIFY.md}. Here $\rho_1(j,K)=\|H_j(K)\|/L$ with $H_j(K)$ the one-step gauge-fixed minimizer map of Section~\ref{sec:H2}.

\begin{theorem}[(H2) at $L=13$, ledger row R-025]\label{thm:L046}
For $L=13$, every level $j\ge1$ including $j=\infty$, and every coarse momentum $K\in\BZ$,
\[
\rho_1(j,K)^2\le10<13 .
\]
Hence $\rho_2\le\sup_{j,K}\rho_1(j,K)^2\le10<L$ by the factorization $\rho_2\le\rho_1(j)\max_l\rho_1(j-1,k_l)$ (Lean \texttt{rho2\_le}, Section~\ref{sec:onestep}). That is, (H2) (ledger row R-025) holds at $L=13$.
\end{theorem}

\emph{Proof sketch.} Write $s=L^{-2}$, $k_l=(K+2\pi l)/L$, $M=L^{j-1}$, $d^M$ for the Prop.\ A symbol at level $M$, $P_\perp$ for the projection onto $\hat K^\perp$, and $T$, $T_2$, $\sigma$, $Z=(P_\perp TP_\perp)^+$ as in \file{H1_ANALYTIC} \S1.
\begin{enumerate}
\item Exact split. $\rho_1(j,K)^2L^2=\lambda_{\max}\bigl(ZT_2Z+cc^\dagger/(\sigma|\hat K|^4)\bigr)$, $c^\dagger=\hat K^\dagger(1-TZ)$: a transverse part and an orthogonal pure-gauge part.
\item Three exact lemmas. L-264 (Lemma S): the averaging symbol is $E_q(k_l)=E^0_l-\hat Kg_l^\dagger$, with $E^0_l$ diagonal (block average times straight path), $|E^0_{l,\mu\mu}|$ built from $a_{l,\rho}=\sin(K_\rho/2)/(L\sin(k_{l,\rho}/2))$, and $\sum_l\prod_\rho a_{l,\rho}^2=1$ (Fej\'er). Tower identity: $d^{ML}_\mu(K)=L^{-2}\sum_lw_{l,\mu}d^M_\mu(k_l)$, hence $P_\perp TP_\perp=P_\perp\operatorname{diag}(d^{ML}(K))P_\perp$ and $Z$ is the next-level kernel of Prop.\ A. L-166 (Lemma G): a closed form of the gauge row $g_l$ from the path average, so the averaging operator is explicit.
\item Majorant. L-283 (Lemma T) (transverse): $\lambda_{\max}(ZT_2Z)/L^2\le\gamma_LF(K)$, $F(K)=1/(|K|^2\underline d(K))$, $\gamma_L=\operatorname{sinc}(\pi/2L)^{-2}$, from $T_2\le d_{\max}T$, $ZTZ=Z$, $d_{\max}\le L^2\gamma_L/|K|^2$, and $\|Z\|\le1/\underline d$. Lemma Gauge (L-180): the gauge part is $\le(1+\tau)^2/(|\hat K|^2S_L)$ with explicit sums $V$, $U$ and $\tau=(V+|\hat K|U)/\underline d$. So $\rho_1(j,K)^2\le B_L(K):=\gamma_LF(K)+(1+\tau)^2/(|\hat K|^2S_L)$. By Cauchy--Schwarz $F\le F_{CS}$, a closed form made of one-dimensional factors in each $K_\rho$.
\item Uniformity in $j$. $B_L$ does not depend on $j$. For every $M\in\{1,L,L^2,\dots,\infty\}$ the fine symbol satisfies $d^M\le1/|\tilde q^M_0|^2\le1/|\hat k|^2$, because the Fej\'er weights sum to 1. The coarse symbol $d^{ML}$ dominates the truncated $M=\infty$ sum termwise. No monotonicity in $j$ and no O5 rate is used.
\item Branch-and-bound. $B_L$ is invariant under the 384 signed permutations of $K$ by construction, so the sup over the sorted chamber $0\le K_1\le\dots\le K_4\le\pi$ is the sup over $\BZ$. The program \file{numerics/h1analytic/bb_arb.py} encloses $\gamma_LF_{CS}+\bar G_L$ on boxes $K=\pi u$ with exact dyadic $u$, evaluates every one-dimensional value in arb (python-flint), and rounds outward by one ulp after every floating-point operation, with Higham bounds on every sum. Result: $\sup_{\text{chamber}}B_{13}\le9.9995277$, 25{,}169 boxes, 8{,}075 gauge evaluations, 0 failed leaves, 58\,s. Boxes that contain $K=0$ are enclosed directly; $\rho_1(j,0)=1$.
\end{enumerate}
Margins. The measured $\sup\rho_1^2$ is 4.37. The majorant is tight to 2.6\% at $|K|\approx1.5$ ($\min B_{13}/\rho_1^2=1.026$ over 896 cases and a local search). The certificate refines boxes only until each is $\le10$, so its margin to 10 is $5\cdot10^{-5}$; its margin to the threshold 13 is 23\%.

Verification. \file{H1_ANALYTIC_VERIFY} rederives the split, Lemmas S, G, the tower identity, L-283 (Lemma T), and Lemma Gauge on paper and checks them against the production objects to $\le5.3\cdot10^{-13}$ (V1--V2); it checks $B_{13}\ge\rho_1^2$ at 896 $(j,K)$ cases (V3--V4); it grades the original float program CORRECTED (no directed rounding) and replaces it by \texttt{bb\_arb.py} (V5--V8). \file{H1_ANALYTIC_VERIFY2} rebuilds $H_j(K)$ from the definitions (position-space (0.4) paths, sparse KKT, ghost-domain gauge) and reproduces the production $\rho_1$ to $5.8\cdot10^{-9}$ ($10^{-14}$ with the same $d$ symbol); the two parts of the split agree to $3\cdot10^{-8}$, and the L-264 (Lemma S) conventions are derived by hand and confirmed to $1.2\cdot10^{-13}$. \file{H2_CERTIFY_P5_VERIFY} proves and checks the invariance of $\rho_1$ under the 384 signed permutations (P5a, P5b); the analytic certificate does not need it, the cell sweep does. Lean: \file{lean/YmLean/Tower.lean} formalizes the algebraic core of the majorant (tower identity in weighted form, Rayleigh and rank-one bounds, \texttt{majorant\_core}, \texttt{lemmaT\_core}); \texttt{BlockNorm.rho2\_le} is the one-step $\Rightarrow$ two-step factorization.
